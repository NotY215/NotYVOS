cmake_minimum_required(VERSION 3.16)
project(notyvos-xmb CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED sdl2 SDL2_ttf SDL2_image)
add_executable(notyvos-xmb main.cpp)
target_include_directories(notyvos-xmb PRIVATE ${SDL2_INCLUDE_DIRS})
target_link_libraries(notyvos-xmb PRIVATE ${SDL2_LIBRARIES})
target_compile_options(notyvos-xmb PRIVATE ${SDL2_CFLAGS_OTHER} -Wall -Wextra)
install(TARGETS notyvos-xmb RUNTIME DESTINATION bin)
